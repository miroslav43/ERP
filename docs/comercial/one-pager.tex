\newcommand{\sectiune}[1]{%
  \par\vspace{6pt}

% Rând de modul: iconiță, denumire îngroșată, o frază. Același tipar ca
% `\randlista` din `tema/beamer.tex` — casetă de lățime fixă pentru
% iconiță și `\hangindent` egal cu ea, ca a doua linie să cadă sub text.
\newcommand{\modulrand}[3]{%
  \par\hangindent=1.65em\hangafter=1\noindent
  \makebox[1.25em][l]{\textcolor{brandBlue}{\faIcon{#1}}}\hspace{0.4em}%
  \textbf{#2}\,{\color{brandSlate} — #3}\par\vspace{2.2pt}}

% Card de grupă: bara colorată din stânga poartă culoarea grupei, exact
% ca pe harta de module din prezentare.
\newtcolorbox{grupa}[3][brandBlue]{%
  enhanced, arc=3pt,
  colback=white, colframe=brandLine, boxrule=0.5pt,
  borderline west={2.6pt}{0pt}{#1},
  left=7pt,right=7pt,top=3pt,bottom=3pt,
  halign=flush left, halign title=flush left,
  fonttitle=\bfseries\small,
  title={\textcolor{#1}{\faIcon{#2}}\hspace{5pt}\textcolor{brandNavy}{#3}},
  colbacktitle=white, coltitle=brandNavy,
  toptitle=2pt, bottomtitle=1pt}

% Un rol: iconița lângă nume, dedesubt ce vede. Iconița deasupra costa un
% rând întreg, iar pagina trecea pe a doua cu banda de jos.
\newcommand{\rol}[3]{%
  \begin{minipage}[t]{0.24\linewidth}
    {\small\textcolor{brandBlue}{\faIcon{#1}}\hspace{5pt}\bfseries #2}\par\vspace{1pt}
    {\footnotesize\color{brandSlate}#3}
  \end{minipage}}

% O promisiune din banda de jos: iconiță și două rânduri scurte.
\newcommand{\promisiune}[3]{%
  \begin{minipage}[t]{0.19\linewidth}
    \centering
    {\color{brandTeal}\large\faIcon{#1}}\par\vspace{3pt}
    {\footnotesize\bfseries\color{brandNavy}#2}\par
    {\scriptsize\color{brandSlate}#3}
  \end{minipage}}

\sectiune{Ce facem}
{\small
\textbf{Administrativo} este aplicația web în care o firmă din România își administrează
activitatea internă: oamenii, prezența, concediile, salariile, echipamentele și actele
care le leagă. \textbf{O singură evidență} — un angajat introdus o dată apare în pontaj,
în concedii, în salarizare și pe bunurile din primirea lui. Se lucrează din browser, pe
calculator sau pe telefon; nu instalați nimic și nu aveți servere de întreținut.}
