\begin{frame}{Every Romanian employer owes the state the same HR paperwork}
  \vspace{0pt}
  {\footnotesize\color{brandSlate} About half a million companies employ staff. Each one
  must keep a daily time record and report every hire, change and termination to the
  Labour Inspectorate (ITM), with fines counted per employee.}
  \vspace{8pt}

  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.43\textwidth}
      \stat{\linewidth}{≈526,000}{Romanian companies employ at least one person (FY2025),
        with 3.88M staff in total}\par\vspace{8pt}
      \stat{\linewidth}{3,000–8,000 RON}{fine \emph{per person} for late or missing
        REGES-Online data (≈€600–1,600; 5 RON = 1 EUR)}\par\vspace{8pt}
      \stat{\linewidth}{1,500–3,000 RON}{fine for not keeping a daily record of start and
        end hours (Labour Code art.~119 and art.~260)}
    \end{column}
    \begin{column}{0.54\textwidth}
      \begin{card}[brandAmber]{clipboard-list}{What a 10–50-person firm must run, every month}
        \scriptsize
        \pct{clock}{A \textbf{daily start/end record} for every employee. Paper and Excel are
          legal, and both can be rewritten after the fact.}\par\vspace{2pt}
        \pct{file-signature}{\textbf{REGES-Online}: a new contract by the day before work
          starts; other changes within 3–20 working days.}\par\vspace{2pt}
        \pct{calendar-alt}{Leave requests and balances, public holidays, approvals.}\par\vspace{2pt}
        \pct{hard-hat}{Health \& safety (SSM) briefings, medical checks, fire extinguishers,
          vehicle documents: each with its own expiry date.}\par\vspace{2pt}
        \pct{calculator}{Month-end hand-over of hours and leave to the accountant, who runs payroll.}
      \end{card}
    \end{column}
  \end{columns}

  \surse{Sources: ICAP CRIF FY2025 via Curierul Național (Jul 2026); fines: HG 295/2025 art.~9 as
    summarised by pontajangajati.ro and universuljuridic.ro; Labour Code art.~119/260 via lege5.ro and
    ITM Arad; REGES deadlines: Legislația Muncii 2026 guide. Full URLs: Data room › 5 Market
    research, §a.1, §b.1–b.3. REGES-Online: the Labour Inspectorate's mandatory electronic
    employee register.}
\end{frame}
